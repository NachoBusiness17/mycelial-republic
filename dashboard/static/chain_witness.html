import json, hashlib
h = lambda b: hashlib.sha256(b).hexdigest()
rows = [json.loads(l) for l in open('memory/biography/verkle_chain.jsonl',
        encoding='utf-8') if l.strip()]
L = [r['leaf_hash'] for r in rows]
while len(L) > 1:
    L = [h(('node:' + L[i] + ':' + (L[i+1] if i+1 < len(L) else L[i])).encode())
         for i in range(0, len(L), 2)]
print('recomputed :', L[0])
print('published  :', rows[-1]['verkle_root'])
print('MATCH' if L[0] == rows[-1]['verkle_root']
